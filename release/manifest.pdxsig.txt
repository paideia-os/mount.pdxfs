# mount.pdxfs v1.1.1 — release manifest (source form)
#
# This is the human-readable source form of `manifest.pdxsig`. The
# release tool at pkgs.paideia-os mirror-push time (a) enumerates the
# artifact list below, (b) recomputes every BLAKE3-256 hash from the
# repo tree at the v1.1.1 tag, (c) writes the completed manifest into
# a binary `manifest.pdxsig`, and (d) invokes
#   paideia-pq-sign::sign_release_artifact(sk, manifest_path)
# once with the paideia-release-line Ed25519 secret key and once with
# the ML-DSA-65 secret key. The two detached signatures are appended
# to the manifest in the fixed layout below and the resulting
# `manifest.pdxsig` is pushed to
#   https://pkgs.paideia-os/main/mount.pdxfs/1.1.1/manifest.pdxsig
# alongside the compiled artifacts.
#
# Format mirrors `libpdx-volume`'s `release/manifest.pdxsig.txt` (this
# wave's other satellite, first through M5) byte-for-byte in structure
# — see that file for the worked template this one follows. Every
# `<BLAKE3-*>` / `<...-KID-*>` / `<...-SIG-*>` slot below is a
# placeholder the release tool fills in from the working tree at tag
# time; NONE are computed by this milestone.
#
# Reference:
#   design/paideia-as/v0.20-issue-1025-pq-signing.md         — Hybrid,
#     sign_release_artifact, HybridSignature (Ed25519 || ML-DSA-65).
#   design/02-development-environment.md §1140                — hybrid
#     release-artifact signature rationale (Ed25519 + ML-DSA-65).
#   design/tooling/plan.md §6.3                               — package
#     repository layout, `manifest.pdxsig` per version directory.
#   design/tooling/volume-tooling-ux.md §9.2/§9.3 (paideia-os) — the
#     original M1..M5 scope this repo grew out of; M6 (libpdx-volume
#     v1.1.0 API adoption) is a post-R53 cascade tracked in CHANGELOG.md.

#   KERNEL GAP #B — libpdx-volume.snap_chain_walk is itself a stub.
#                   --snapshot-list refuses SNAPSHOT_NOT_IMPL.
#   GAP #P1       — no pdxb_sb_get_dek_nonce accessor in libpdx-volume
#                   v1.1.0. src/passphrase_wire.pdx uses a direct
#                   offset arithmetic workaround. Flagged for v1.1.x.
#   GAP #P2       — no KIND_DEK cap kind kernel-side. Unwrapped DEK
#                   stays in .bss scratch; "install as cap" hop is a
#                   future landing.
#   GAP #Q1       — no KIND_PDXFS_QUOTA cap kind kernel-side. Every
#                   mount with PDXB_FLAG_HAS_QUOTA set silently
#                   proceeds without quota enforcement.

# ---------------------------------------------------------------
# Signature block — dual-signed per design/02-development-environment.md
# §1140 (hybrid Ed25519 + ML-DSA-65). Both signatures cover the SAME
# canonical byte-stream: the manifest body above (everything before
# the [signatures] marker line) hashed with BLAKE3-256 and signed via
# paideia-pq-sign::sign_release_artifact. Verification is AND-semantics:
# both signatures MUST verify; failure of either rejects the package.
#
# NOT PERFORMED AT THIS MILESTONE. The paideia-as `mldsa65_sign`
# intrinsic requires a live 32-byte ML-DSA-65 seed key, and this repo
# holds no such key material — a signing seed is release-line custody,
# never repo-resident. The signature block below is left as the
# documented placeholder value `SIGNATURE_PLACEHOLDER_PENDING_LIVE_SIGN`
# in every slot the real dual-sign pass would fill; see
# release/RELEASE-1.0.0.md §3 for the full rationale and the flag for
# main (the 1.0.0 release note's dual-sign section applies verbatim to
# 1.1.0 — no substrate change to the signing pipeline this release).
# ---------------------------------------------------------------

[signatures]
signer-identity      = paideia-release-line
signer-key-source    = design/02-development-environment.md §1164
                       (hardware-backed in CI — TPM 2.0 on the release
                       runner, or a cloud KMS pending PQ-KMS
                       availability check per that section's TODO).
signature-scheme     = hybrid-ed25519+ml-dsa-65 (paideia-pq-hybrid-v1)
hash-input-algo      = blake3-256
hash-input-domain    = "paideia-release-artifact\x00" || manifest-body-bytes
                       (matches paideia-pq-sign::sign_release_artifact
                       domain separation, crates/paideia-pq-sign/src/
                       release.rs.)

ed25519-key-id       = SIGNATURE_PLACEHOLDER_PENDING_LIVE_SIGN
ed25519-signature    = SIGNATURE_PLACEHOLDER_PENDING_LIVE_SIGN

ml-dsa-65-key-id     = SIGNATURE_PLACEHOLDER_PENDING_LIVE_SIGN
